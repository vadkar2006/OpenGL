// PATCH: safe spring normalize
vec3 Safe_Spring_Deformation(vec3 d, float rest_length) {
    float len = length(d);
    if (len < 1e-8) return vec3(0.0);
    return d - (d / len) * rest_length;
}

void main() {
    // Получаем координаты текущего потока
    ivec3 pixel = ivec3(gl_GlobalInvocationID.xyz);

    bool flag_X = pixel.x <= 0 || pixel.x >= (u_count_xyzt.x - 1);
    bool flag_Y = pixel.y <= 0 || pixel.y >= (u_count_xyzt.y - 1);
    bool flag_Z = pixel.z <= 0 || pixel.z >= (u_count_xyzt.z - 1);

    if(flag_X || flag_Y || flag_Z) return;
    


    vec3 pos_center = Get_History_Pos(pixel.x, pixel.y, pixel.z, 0);
    float k_center = Get_History_Stiffness(pixel.x, pixel.y, pixel.z, 0);
    float m_center = Get_History_Mass(pixel.x, pixel.y, pixel.z, 0);

    //векторы до грузов от центрального
    vec3 pos_x_plus = Get_History_Pos(pixel.x + 1, pixel.y, pixel.z, 0) + vec3(1.0, 0.0, 0.0)*u_dlt.x - pos_center;
    vec3 pos_x_minus = Get_History_Pos(pixel.x - 1, pixel.y, pixel.z, 0) - vec3(1.0, 0.0, 0.0)*u_dlt.x - pos_center;
    vec3 pos_y_plus = Get_History_Pos(pixel.x, pixel.y + 1, pixel.z, 0) + vec3(0.0, 1.0, 0.0)*u_dlt.x - pos_center;
    vec3 pos_y_minus = Get_History_Pos(pixel.x, pixel.y - 1, pixel.z, 0) - vec3(0.0, 1.0, 0.0)*u_dlt.x - pos_center;
    vec3 pos_z_plus = Get_History_Pos(pixel.x, pixel.y, pixel.z + 1, 0) + vec3(0.0, 0.0, 1.0)*u_dlt.x - pos_center;
    vec3 pos_z_minus = Get_History_Pos(pixel.x, pixel.y, pixel.z - 1, 0) - vec3(0.0, 0.0, 1.0)*u_dlt.x - pos_center;

    pos_x_plus = Safe_Spring_Deformation(pos_x_plus, u_dlt.x);
    pos_x_minus = Safe_Spring_Deformation(pos_x_minus, u_dlt.x);
    pos_y_plus = Safe_Spring_Deformation(pos_y_plus, u_dlt.x);
    pos_y_minus = Safe_Spring_Deformation(pos_y_minus, u_dlt.x);
    pos_z_plus = Safe_Spring_Deformation(pos_z_plus, u_dlt.x);
    pos_z_minus = Safe_Spring_Deformation(pos_z_minus, u_dlt.x);
    
    //жесткости
    float k_x_plus = 1 / (1 / k_center  +  1 / Get_History_Stiffness(pixel.x + 1, pixel.y, pixel.z, 0)); 
    float k_x_minus = 1 / (1 / k_center  +  1 / Get_History_Stiffness(pixel.x - 1, pixel.y, pixel.z, 0)); 
    float k_y_plus = 1 / (1 / k_center  +  1 / Get_History_Stiffness(pixel.x, pixel.y + 1, pixel.z, 0)); 
    float k_y_minus = 1 / (1 / k_center  +  1 / Get_History_Stiffness(pixel.x, pixel.y - 1, pixel.z, 0)); 
    float k_z_plus = 1 / (1 / k_center  +  1 / Get_History_Stiffness(pixel.x, pixel.y, pixel.z + 1, 0)); 
    float k_z_minus = 1 / (1 / k_center  +  1 / Get_History_Stiffness(pixel.x, pixel.y, pixel.z - 1, 0));

    //итоговое ускорение
    vec3 a_center = (pos_x_plus*k_x_plus + pos_x_minus*k_x_minus + pos_y_plus*k_y_plus + pos_y_minus*k_y_minus + pos_z_plus*k_z_plus + pos_z_minus*k_z_minus) / m_center;

    vec3 v_center = Get_History_Speed(pixel.x, pixel.y, pixel.z, 0) + a_center*u_dlt.y;
    vec3 x_center = pos_center + v_center*u_dlt.y;


    
    Set_History_Pos(pixel.x, pixel.y, pixel.z, 1, x_center);
    Set_History_Speed(pixel.x, pixel.y, pixel.z, 1, v_center);
    Set_History_Acceleration(pixel.x, pixel.y, pixel.z, 1, a_center);
}

uniform int u_z_slice; 

uniform vec3 u_dxyz;

uniform ivec3 u_xyz_count;

uniform vec3 position_zoom;



void main() {
    vec2 ndc = (TexCoord - vec2(0.5, 0.5)) * 2.0;


    ndc = ndc * position_zoom.z + position_zoom.xy;


    vec2 uv = ndc * 0.5 + 0.5;

    if (u_z_slice < 0 || u_z_slice >= u_xyz_count.z) {
        FragColor = vec4(0.5, 0.0, 0.5, 1.0);
        return;
    }

    if (uv.x < 0.0 || uv.x > 1.0 || uv.y < 0.0 || uv.y > 1.0) {
        FragColor = vec4(0.08, 0.08, 0.08, 1.0);
        return;
    }

    vec2 grid_f = uv * vec2(float(u_xyz_count.x - 1), float(u_xyz_count.y - 1));
    ivec2 cell = ivec2(floor(grid_f));

    if (cell.x < 0 || cell.x >= u_xyz_count.x || cell.y < 0 || cell.y >= u_xyz_count.y) {
        FragColor = vec4(0.08, 0.08, 0.08, 1.0);
        return;
    }

    int index = (u_current_step * u_xyz_count.x * u_xyz_count.y * u_xyz_count.z) +
                (u_z_slice * u_xyz_count.x * u_xyz_count.y) +
                (cell.y * u_xyz_count.x) +
                cell.x;

    vec3 p = history_pos[index];

    if (isnan(p.x) || isnan(p.y) || isnan(p.z)) {
        FragColor = vec4(1.0, 0.0, 1.0, 1.0);
        return;
    }

    float mag = abs(p.x) + abs(p.y) + abs(p.z);


    if (mag < 1e-7) {
        float checker = (mod(float(cell.x + cell.y), 2.0) < 1.0) ? 0.12 : 0.18;
        FragColor = vec4(vec3(checker), 1.0);
        return;
    }


    vec3 col = vec3(0.5) + p * 50.0;
    FragColor = vec4(clamp(col, vec3(0.0), vec3(1.0)), 1.0);
}

std::string readShaderFile(const std::string& filepath) {
    std::ifstream file(filepath);
    if (!file.is_open()) {
        std::cerr << "Failed to open shader file: " << filepath << std::endl;
        return "";
    }
    
    std::stringstream buffer;
    buffer << file.rdbuf();
    return buffer.str();
}

GLuint Create_Shader_Program() {
    // Определяем пути к шейдерам (относительно рабочей директории)
    std::string vertexPath = "../src/Shaders/Vertex_Shader.glsl";
    std::string fragmentPath = "../src/Shaders/Fragment_Shader.glsl";
    
    // Читаем шейдеры из файлов
    std::string vertexCode = readShaderFile(vertexPath);
    std::string fragmentCode = readShaderFile(fragmentPath);
    
    if (vertexCode.empty() || fragmentCode.empty()) {
        std::cerr << "Failed to read shader files!" << std::endl;
        return 0;
    }
    
    // Компилируем вершинный шейдер
    GLuint vertexShader = glCreateShader(GL_VERTEX_SHADER);
    const char* vertexSource = vertexCode.c_str();
    glShaderSource(vertexShader, 1, &vertexSource, NULL);
    glCompileShader(vertexShader);
    
    int success;
    char infoLog[512];
    glGetShaderiv(vertexShader, GL_COMPILE_STATUS, &success);
    if (!success) {
        glGetShaderInfoLog(vertexShader, 512, NULL, infoLog);
        std::cerr << "Vertex Shader Error:\n" << infoLog << std::endl;
        return 0;
    }
    
    // Компилируем фрагментный шейдер
    GLuint fragmentShader = glCreateShader(GL_FRAGMENT_SHADER);
    const char* fragmentSource = fragmentCode.c_str();
    glShaderSource(fragmentShader, 1, &fragmentSource, NULL);
    glCompileShader(fragmentShader);
    
    glGetShaderiv(fragmentShader, GL_COMPILE_STATUS, &success);
    if (!success) {
        glGetShaderInfoLog(fragmentShader, 512, NULL, infoLog);
        std::cerr << "Fragment Shader Error:\n" << infoLog << std::endl;
        return 0;
    }
    
    // Создаем программу
    GLuint shaderProgram = glCreateProgram();
    glAttachShader(shaderProgram, vertexShader);
    glAttachShader(shaderProgram, fragmentShader);
    glLinkProgram(shaderProgram);
    
    glGetProgramiv(shaderProgram, GL_LINK_STATUS, &success);
    if (!success) {
        glGetProgramInfoLog(shaderProgram, 512, NULL, infoLog);
        std::cerr << "Shader Program Error:\n" << infoLog << std::endl;
        return 0;
    }
    
    glDeleteShader(vertexShader);
    glDeleteShader(fragmentShader);
    
    return shaderProgram;
}

void main() {
    vec2 ndc = (TexCoord - vec2(0.5, 0.5)) * 2.0;

    // Zoom and pan
    ndc = ndc * position_zoom.z + position_zoom.xy;

    // Convert back to [0..1]
    vec2 uv = ndc * 0.5 + 0.5;

    // Keep aspect without moving the visible grid outside [0..1]
    float aspect = float(window.x) / max(float(window.y), 1.0);
    if (aspect > 1.0) {
        uv.x = (uv.x - 0.5) / aspect + 0.5;
    } else {
        uv.y = (uv.y - 0.5) * aspect + 0.5;
    }

    if (u_z_slice < 0 || u_z_slice >= u_xyz_count.z) {
        FragColor = vec4(0.1, 0.1, 0.1, 1.0);
        return;
    }

    if (uv.x < 0.0 || uv.x > 1.0 || uv.y < 0.0 || uv.y > 1.0) {
        FragColor = vec4(0.1, 0.1, 0.1, 1.0);
        return;
    }

    ivec2 cell = ivec2(floor(uv * vec2(float(u_xyz_count.x - 1), float(u_xyz_count.y - 1))));

    if (cell.x < 0 || cell.x >= u_xyz_count.x || cell.y < 0 || cell.y >= u_xyz_count.y) {
        FragColor = vec4(0.1, 0.1, 0.1, 1.0);
        return;
    }

    int index = (u_current_step * u_xyz_count.x * u_xyz_count.y * u_xyz_count.z) +

    GLFWwindow* window = Start_Window();

    if (shaderProgram == 0) return -1;

    // Данные для полноэкранного квада
    float vertices[] = {
        // positions      // texcoords
        -1.0f,  1.0f,     0.0f, 1.0f, // top left
        -1.0f, -1.0f,     0.0f, 0.0f, // bottom left
         1.0f, -1.0f,     1.0f, 0.0f, // bottom right
         1.0f,  1.0f,     1.0f, 1.0f  // top right
    };
    unsigned int indices[] = {
        0, 1, 2,
        2, 3, 0
    };

    GLuint VAO, VBO, EBO;
    VAO_VBO_EBO(VAO, VBO, EBO, vertices, indices, sizeof(vertices), sizeof(indices));



    Grid_Solving grid_temp(500, 500, 3, 3, 0.01, 0.0001);
    

    grid_temp.Get_Grid(0)[grid_temp(true, 250, 250, 1, 0)] = 0.1;


    grid_temp.Compilation_VideoCard_Program("Compute_Shader.glsl");

    grid_temp.Go_To_VideoCard();
    glMemoryBarrier(GL_ALL_BARRIER_BITS); // PATCH: make initial CPU upload visible to GPU shaders



    // --- Регистрация колбэков ввода ---
    glfwSetKeyCallback(window, keyCallback);
    glfwSetMouseButtonCallback(window, mouseButtonCallback);





    for(int i = 0; i < Grid_Solving::COUNT_SSBO; i++)   
        glBindBufferBase(GL_SHADER_STORAGE_BUFFER, i, grid_temp.Get_ssbo(i));

    GLuint u_count_xyzt = glGetUniformLocation(grid_temp.Get_VideoCard_Program(), "u_count_xyzt");
    GLuint u_dlt = glGetUniformLocation(grid_temp.Get_VideoCard_Program(), "u_dlt");
    GLuint u_current_step = glGetUniformLocation(grid_temp.Get_VideoCard_Program(), "u_current_step");





    GLuint render_u_z_slice = glGetUniformLocation(shaderProgram, "u_z_slice");
    GLuint render_u_current_step = glGetUniformLocation(shaderProgram, "u_current_step");

    GLuint render_u_count_xyz = glGetUniformLocation(shaderProgram, "u_xyz_count");
    GLuint render_u_dxyz = glGetUniformLocation(shaderProgram, "u_dxyz");

    GLuint render_window = glGetUniformLocation(shaderProgram, "window");

    GLuint render_position_zoom = glGetUniformLocation(shaderProgram, "position_zoom");


    
    double lastTime = glfwGetTime();
    int frameCount = 0;
    int total_step = 0;
    while (!glfwWindowShouldClose(window)) {
        if (glfwGetKey(window, GLFW_KEY_W) == GLFW_PRESS) y_pos += 0.001 * zoom;
        if (glfwGetKey(window, GLFW_KEY_S) == GLFW_PRESS) y_pos -= 0.001 * zoom;
        if (glfwGetKey(window, GLFW_KEY_D) == GLFW_PRESS) x_pos += 0.001 * zoom;
        if (glfwGetKey(window, GLFW_KEY_A) == GLFW_PRESS) x_pos -= 0.001 * zoom;

        if(glfwGetMouseButton(window, GLFW_MOUSE_BUTTON_1) == GLFW_PRESS) zoom /= 1.003; 
        if(glfwGetMouseButton(window, GLFW_MOUSE_BUTTON_2) == GLFW_PRESS) zoom *= 1.003; 


        glUseProgram(grid_temp.Get_VideoCard_Program());
        
        
        
        glUniform4i(u_count_xyzt, grid_temp.Get_Count_Nodes_X(), grid_temp.Get_Count_Nodes_Y(), grid_temp.Get_Count_Nodes_Z(), grid_temp.Get_Count_Nodes_T());
        glUniform2f(u_dlt, grid_temp.Get_dl(), grid_temp.Get_dt());
        
        int groupsX = (grid_temp.Get_Count_Nodes_X() + 15) / 16;
        int groupsY = (grid_temp.Get_Count_Nodes_Y() + 15) / 16;
        int groupsZ = (grid_temp.Get_Count_Nodes_Z() + 0) / 1;
        
        for (int i = 0; i < frame_count_go_program; i++) {
            glUniform1i(u_current_step, total_step % grid_temp.Get_Count_Nodes_T()); // PATCH: ensure compute step is inside time ring buffer
            
            glDispatchCompute(groupsX, groupsY, groupsZ);
            
            glMemoryBarrier(GL_SHADER_STORAGE_BARRIER_BIT);
            
            total_step = (total_step + 1) % grid_temp.Get_Count_Nodes_T();
        }


        





        // ОТРИСОВКА (Render Shader) --------------------------------------------------------------------------------------  
        glUseProgram(shaderProgram);


        glUniform1i(render_u_z_slice, 1); // Смотрим срез Z=1, где вы поставили 1000
        glUniform1i(render_u_current_step, total_step % grid_temp.Get_Count_Nodes_T()); // PATCH: ensure render step is inside time ring buffer
        glUniform2i(render_window, SCR_WIDTH, SCR_HEIGHT);
        glUniform3i(render_u_count_xyz, grid_temp.Get_Count_Nodes_X(), grid_temp.Get_Count_Nodes_Y(), grid_temp.Get_Count_Nodes_Z());

        glUniform3f(render_u_dxyz, grid_temp.Get_dl(), grid_temp.Get_dl(), grid_temp.Get_dl());
        glUniform3f(render_position_zoom, x_pos, y_pos, zoom);


        glBindVertexArray(VAO);
        glDrawElements(GL_TRIANGLES, 6, GL_UNSIGNED_INT, 0);

        glfwSwapBuffers(window);
        



        // 3. Подсчет FPS
        frameCount++;
        double currentTime = glfwGetTime();
        if (currentTime - lastTime >= 5.0) {
            double fps = frameCount / (currentTime - lastTime);
            
            std::cout << "FPS: " << fps << std::endl;

            frameCount = 0;
            lastTime = currentTime;
        }
        
        
        glfwPollEvents();
    }


    glDeleteVertexArrays(1, &VAO);
    glDeleteBuffers(1, &VBO);
    glDeleteBuffers(1, &EBO);
    glDeleteProgram(shaderProgram);
    glfwDestroyWindow(window);
    glfwTerminate();
    return 0;
}

void VAO_VBO_EBO(GLuint& VAO, GLuint& VBO, GLuint& EBO, float vertices[], unsigned int indices[], unsigned int sizeof_vertices, unsigned int sizeof_indices){
    // Создаем VAO
    glGenVertexArrays(1, &VAO);
    glBindVertexArray(VAO);
    
    // Создаем VBO и загружаем вершины
    glGenBuffers(1, &VBO);
    glBindBuffer(GL_ARRAY_BUFFER, VBO);
    glBufferData(GL_ARRAY_BUFFER, sizeof_vertices, vertices, GL_STATIC_DRAW);
    
    // Создаем EBO и загружаем индексы
    glGenBuffers(1, &EBO);
    glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, EBO);
    glBufferData(GL_ELEMENT_ARRAY_BUFFER, sizeof_indices, indices, GL_STATIC_DRAW);
    
    // Настраиваем атрибуты вершин
    // Атрибут 0: позиция (2 float)
    glVertexAttribPointer(0, 2, GL_FLOAT, GL_FALSE, 4 * sizeof(float), (void*)0);
    glEnableVertexAttribArray(0);
    
    // Атрибут 1: текстурные координаты (2 float)
    glVertexAttribPointer(1, 2, GL_FLOAT, GL_FALSE, 4 * sizeof(float), (void*)(2 * sizeof(float)));
    glEnableVertexAttribArray(1);
}

    glClearColor(0.02f, 0.02f, 0.02f, 1.0f); // PATCH v2: clear color
    while (!glfwWindowShouldClose(window)) {
        if (glfwGetKey(window, GLFW_KEY_W) == GLFW_PRESS) y_pos += 0.001 * zoom;
        if (glfwGetKey(window, GLFW_KEY_S) == GLFW_PRESS) y_pos -= 0.001 * zoom;
        if (glfwGetKey(window, GLFW_KEY_D) == GLFW_PRESS) x_pos += 0.001 * zoom;
        if (glfwGetKey(window, GLFW_KEY_A) == GLFW_PRESS) x_pos -= 0.001 * zoom;

        if(glfwGetMouseButton(window, GLFW_MOUSE_BUTTON_1) == GLFW_PRESS) zoom /= 1.003; 
        if(glfwGetMouseButton(window, GLFW_MOUSE_BUTTON_2) == GLFW_PRESS) zoom *= 1.003; 


        glUseProgram(grid_temp.Get_VideoCard_Program());
        
        
        
        glUniform4i(u_count_xyzt, grid_temp.Get_Count_Nodes_X(), grid_temp.Get_Count_Nodes_Y(), grid_temp.Get_Count_Nodes_Z(), grid_temp.Get_Count_Nodes_T());
        glUniform2f(u_dlt, grid_temp.Get_dl(), grid_temp.Get_dt());
        
        int groupsX = (grid_temp.Get_Count_Nodes_X() + 15) / 16;
        int groupsY = (grid_temp.Get_Count_Nodes_Y() + 15) / 16;
        int groupsZ = (grid_temp.Get_Count_Nodes_Z() + 0) / 1;
        
        for (int i = 0; i < frame_count_go_program; i++) {
            glUniform1i(u_current_step, total_step % grid_temp.Get_Count_Nodes_T()); // PATCH: ensure compute step is inside time ring buffer
            
            glDispatchCompute(groupsX, groupsY, groupsZ);
            
            glMemoryBarrier(GL_SHADER_STORAGE_BARRIER_BIT);
            
            total_step = (total_step + 1) % grid_temp.Get_Count_Nodes_T();
        }


        





        // ОТРИСОВКА (Render Shader) --------------------------------------------------------------------------------------  
        glViewport(0, 0, SCR_WIDTH, SCR_HEIGHT); // PATCH v2: clear and viewport
        glClear(GL_COLOR_BUFFER_BIT);
        glUseProgram(shaderProgram);


        glUniform1i(render_u_z_slice, 1); // Смотрим срез Z=1, где вы поставили 1000
        glUniform1i(render_u_current_step, total_step % grid_temp.Get_Count_Nodes_T()); // PATCH: ensure render step is inside time ring buffer
        glUniform2i(render_window, SCR_WIDTH, SCR_HEIGHT);
        glUniform3i(render_u_count_xyz, grid_temp.Get_Count_Nodes_X(), grid_temp.Get_Count_Nodes_Y(), grid_temp.Get_Count_Nodes_Z());

        glUniform3f(render_u_dxyz, grid_temp.Get_dl(), grid_temp.Get_dl(), grid_temp.Get_dl());
        glUniform3f(render_position_zoom, x_pos, y_pos, zoom);


        glBindVertexArray(VAO);
        glDrawElements(GL_TRIANGLES, 6, GL_UNSIGNED_INT, 0);

        glfwSwapBuffers(window);
        



        // 3. Подсчет FPS
        frameCount++;
        double currentTime = glfwGetTime();
        if (currentTime - lastTime >= 5.0) {
            double fps = frameCount / (currentTime - lastTime);
            
            std::cout << "FPS: " << fps << std::endl;

            frameCount = 0;
            lastTime = currentTime;
        }
        
        
        glfwPollEvents();
    }


    glDeleteVertexArrays(1, &VAO);
    glDeleteBuffers(1, &VBO);
    glDeleteBuffers(1, &EBO);
    glDeleteProgram(shaderProgram);
    glfwDestroyWindow(window);
    glfwTerminate();
    return 0;
}
